    COMPLETED = "completed"
    ERROR = "error"


class EntityType(str, Enum):
    DIAGNOSIS = "diagnosis"
    PROCEDURE = "procedure"
    MEDICATION = "medication"
    SYMPTOM = "symptom"
    ALLERGY = "allergy"

    entity_type: EntityType

    end: int

"""WhisperX-based ASR pipeline with medical fine-tuning support."""
import whisperx
import torch
import numpy as np
from typing import Dict, Any, BinaryIO
import tempfile
import os
import soundfile as sf
from app.core.config import settings
import structlog
import asyncio
from concurrent.futures import ThreadPoolExecutor

logger = structlog.get_logger()


class ASRPipeline:
    """
    Production ASR pipeline using WhisperX.
    Supports diarization (speaker identification) and word-level timestamps.
    """
    
    def __init__(self):
        self.device = settings.WHISPER_DEVICE
        self.compute_type = settings.WHISPER_COMPUTE_TYPE
        self.model = None
        self.align_model = None
        self.diarize_model = None
        self._executor = ThreadPoolExecutor(max_workers=2)
        self._load_models()
    
    def _load_models(self):
        """Load WhisperX models - call once at startup."""
        logger.info(
            "loading_whisper_model",
            model=settings.WHISPER_MODEL,
            device=self.device
        )
        
        # 1. Load transcription model
        self.model = whisperx.load_model(
            settings.WHISPER_MODEL,
            self.device,
            compute_type=self.compute_type,
            language=settings.WHISPER_LANGUAGE,
        )
        
        # 2. Load alignment model (for word-level timestamps)
        self.align_model, self.align_metadata = whisperx.load_align_model(
            language_code=settings.WHISPER_LANGUAGE,
            device=self.device
        )
        
        # 3. Load diarization model (optional - identifies speakers)
        try:
            self.diarize_model = whisperx.DiarizationPipeline(
                model_name="pyannote/speaker-diarization-3.1",
                use_auth_token=None,  # Set HF token if needed
                device=self.device
            )
        except Exception as e:
            logger.warning("diarization_model_load_failed", error=str(e))
            self.diarize_model = None
    
    async def transcribe(
        self, 
        audio_data: bytes,
        sample_rate: int = 16000
    ) -> Dict[str, Any]:
        """
        Transcribe audio bytes to text with medical context.
        
        Args:
            audio_data: Raw audio bytes
            sample_rate: Audio sample rate
            
        Returns:
            Dict with transcript, segments, and metadata
        """
        loop = asyncio.get_event_loop()
        return await loop.run_in_executor(
            self._executor,
            self._sync_transcribe,
            audio_data,
            sample_rate
        )
    
    def _sync_transcribe(
        self, 
        audio_data: bytes, 
        sample_rate: int
    ) -> Dict[str, Any]:
        """Synchronous transcription (runs in thread pool)."""
        
        # Convert bytes to numpy array
        with tempfile.NamedTemporaryFile(suffix=".wav", delete=False) as tmp:
            tmp_path = tmp.name
            # Write proper WAV header if needed
            audio_array = np.frombuffer(audio_data, dtype=np.int16)
            audio_float = audio_array.astype(np.float32) / 32768.0
            sf.write(tmp_path, audio_float, sample_rate)
        
        try:
            # 1. Transcribe with Whisper
            audio = whisperx.load_audio(tmp_path)
            result = self.model.transcribe(
                audio,
                batch_size=settings.WHISPER_BATCH_SIZE,
                language=settings.WHISPER_LANGUAGE
            )
            
            # 2. Align words (precise timestamps)
            result = whisperx.align(
                result["segments"],
                self.align_model,
                self.align_metadata,
                audio,
                self.device,
                return_char_alignments=False
            )
            
            # 3. Diarize (identify speakers - doctor vs patient)
            if self.diarize_model:
                diarize_segments = self.diarize_model(audio)
                result = whisperx.assign_word_speakers(
                    diarize_segments, 
                    result
                )
            
            # 4. Post-process for medical terminology
            transcript = self._post_process_medical(result["segments"])
            
            return {
                "transcript": transcript,
                "segments": result["segments"],
                "language": result.get("language", "en"),
                "confidence": self._calculate_confidence(result["segments"]),
                "speakers": list(set(
                    seg.get("speaker", "UNKNOWN") 
                    for seg in result["segments"]
                ))
            }
            
        finally:
            os.unlink(tmp_path)
    
    def _post_process_medical(self, segments: list) -> str:
        """
        Post-process transcript for medical accuracy.
        - Expand abbreviations
        - Normalize medication names
        - Format numbers (doses, frequencies)
        """
        full_text = " ".join(seg["text"].strip() for seg in segments)
        
        # Common medical expansions
        replacements = {
            "b i d": "twice daily",
            "t i d": "three times daily",
            "q d": "once daily",
            "q h s": "at bedtime",
            "p o": "by mouth",
            "i m": "intramuscular",
            "i v": "intravenous",
            "s c": "subcutaneous",
            "c s": "C-section",
            "c o p d": "COPD",
            "c h f": "CHF",
        }
        
        text_lower = full_text.lower()
        for abbrev, expansion in replacements.items():
            text_lower = text_lower.replace(abbrev, expansion)
        
        return text_lower
    
    def _calculate_confidence(self, segments: list) -> float:
        """Calculate average confidence score."""
        if not segments:
            return 0.0
        
        confidences = []
        for seg in segments:
            if "words" in seg:
                word_conf = [
                    w.get("score", 0.0) 
                    for w in seg["words"] 
                    if "score" in w
                ]
                if word_conf:
                    confidences.extend(word_conf)
        
        return float(np.mean(confidences)) if confidences else 0.85

    """Get or create ASR pipeline singleton."""
    global _asr_pipeline
    if _asr_pipeline is None:
        _asr_pipeline = ASRPipeline()
    return _asr_pipeline

"""Clinical NLP pipeline for entity extraction using BioClinicalBERT."""
import spacy
import scispacy
from transformers import AutoTokenizer, AutoModelForTokenClassification
from transformers import pipeline
from typing import List, Dict, Any, Optional
import re
from app.models.schemas import ClinicalEntity, EntityType
from app.core.config import settings
import structlog
import asyncio
from concurrent.futures import ThreadPoolExecutor

logger = structlog.get_logger()


class ClinicalNLPPipeline:
    """
    Extracts clinical entities using:
    1. scispaCy for medical NER
    2. BioClinicalBERT for diagnosis coding
    3. Med7 for medication extraction
    4. Rule-based procedure detection
    """
    
    def __init__(self):
        self.nlp = None
        self.diagnosis_pipeline = None
        self.med7_nlp = None
        self._executor = ThreadPoolExecutor(max_workers=2)
        self._load_models()
    
    def _load_models(self):
        """Load all NLP models."""
        logger.info("loading_nlp_models")
        
        # 1. scispaCy for general biomedical NER
        try:
            self.nlp = spacy.load("en_core_sci_sm")
            logger.info("scispacy_model_loaded")
        except Exception as e:
            logger.error("scispacy_load_failed", error=str(e))
            self.nlp = spacy.load("en_core_web_sm")
        
        # 2. BioClinicalBERT for diagnosis extraction
        try:
            self.diagnosis_pipeline = pipeline(
                "token-classification",
                model="samrawal/bert-base-uncased_clinical-ner",
                aggregation_strategy="simple",
                device=0 if settings.WHISPER_DEVICE == "cuda" else -1
            )
            logger.info("bioclinicalbert_loaded")
        except Exception as e:
            logger.error("bioclinicalbert_load_failed", error=str(e))
            self.diagnosis_pipeline = None
        
        # 3. Med7 for medications
        try:
            self.med7_nlp = spacy.load("en_core_med7_lg")
            logger.info("med7_loaded")
        except Exception as e:
            logger.warning("med7_load_failed", error=str(e))
            self.med7_nlp = None
    
    async def extract_entities(self, text: str) -> List[ClinicalEntity]:
        """
        Extract all clinical entities from text.
        
        Returns list of entities with standardized codes.
        """
        loop = asyncio.get_event_loop()
        return await loop.run_in_executor(
            self._executor,
            self._sync_extract_entities,
            text
        )
    
    def _sync_extract_entities(self, text: str) -> List[ClinicalEntity]:
        """Synchronous entity extraction."""
        entities = []
        
        # 1. Extract diagnoses using BioClinicalBERT
        diagnoses = self._extract_diagnoses(text)
        entities.extend(diagnoses)
        
        # 2. Extract medications using Med7
        medications = self._extract_medications(text)
        entities.extend(medications)
        
        # 3. Extract procedures using rules
        procedures = self._extract_procedures(text)
        entities.extend(procedures)
        
        # 4. Extract symptoms using scispaCy
        symptoms = self._extract_symptoms(text)
        entities.extend(symptoms)
        
        # Deduplicate by position
        seen = set()
        unique_entities = []
        for ent in sorted(entities, key=lambda x: x.start):
            key = (ent.start, ent.end)
            if key not in seen:
                seen.add(key)
                unique_entities.append(ent)
        
        return unique_entities
    
    def _extract_diagnoses(self, text: str) -> List[ClinicalEntity]:
        """Extract diagnoses and map to ICD-10."""
        entities = []
        
        if self.diagnosis_pipeline:
            results = self.diagnosis_pipeline(text)
            for res in results:
                if res["entity_group"] in ["PROBLEM", "DIAGNOSIS"]:
                    entity = ClinicalEntity(
                        text=res["word"],
                        entity_type=EntityType.DIAGNOSIS,
                        start=res["start"],
                        end=res["end"],
                        confidence=res["score"],
                        icd10=self._map_to_icd10(res["word"])
                    )
                    entities.append(entity)
        
        # Fallback: Common diagnosis patterns
        patterns = [
            (r"\b(diabetes|type\s+\d\s+diabetes|diabetic)\b", "E11.9"),
            (r"\b(hypertension|high\s+blood\s+pressure)\b", "I10"),
            (r"\b(asthma)\b", "J45.9"),
            (r"\b(depression|major\s+depressive)\b", "F32.9"),
            (r"\b(anxiety|generalized\s+anxiety)\b", "F41.1"),
            (r"\b(copd|chronic\s+obstructive)\b", "J44.9"),
            (r"\b(chf|congestive\s+heart\s+failure)\b", "I50.9"),
            (r"\b(osteoarthritis|oa)\b", "M19.90"),
        ]
        
        for pattern, default_icd in patterns:
            for match in re.finditer(pattern, text, re.IGNORECASE):
                # Check if not already extracted
                if not any(e.start == match.start() for e in entities):
                    entities.append(ClinicalEntity(
                        text=match.group(),
                        entity_type=EntityType.DIAGNOSIS,
                        start=match.start(),
                        end=match.end(),
                        confidence=0.8,
                        icd10=default_icd
                    ))
        
        return entities
    
    def _extract_medications(self, text: str) -> List[ClinicalEntity]:
        """Extract medications using Med7."""
        entities = []
        
        if self.med7_nlp:
            doc = self.med7_nlp(text)
            for ent in doc.ents:
                if ent.label_ in ["DRUG", "MEDICATION"]:
                    entities.append(ClinicalEntity(
                        text=ent.text,
                        entity_type=EntityType.MEDICATION,
                        start=ent.start_char,
                        end=ent.end_char,
                        confidence=0.9,
                        rxnorm=self._map_to_rxnorm(ent.text)
                    ))
        
        return entities
    
    def _extract_procedures(self, text: str) -> List[ClinicalEntity]:
        """Extract procedures using keyword matching."""
        entities = []
        
        procedure_keywords = {
            "mri": ("73706", "Imaging"),
            "ct scan": ("74150", "Imaging"),
            "x-ray": ("73060", "Imaging"),
            "ultrasound": ("76700", "Imaging"),
            "biopsy": ("11100", "Procedure"),
            "surgery": ("44970", "Surgery"),
            "laparoscopy": ("49320", "Surgery"),
            "colonoscopy": ("45378", "Procedure"),
            "endoscopy": ("43235", "Procedure"),
            "echocardiogram": ("93306", "Cardiology"),
            "eeg": ("95816", "Neurology"),
            "ekg": ("93000", "Cardiology"),
        }
        
        for keyword, (cpt, category) in procedure_keywords.items():
            for match in re.finditer(rf"\b{keyword}\w*\b", text, re.IGNORECASE):
                entities.append(ClinicalEntity(
                    text=match.group(),
                    entity_type=EntityType.PROCEDURE,
                    start=match.start(),
                    end=match.end(),
                    confidence=0.85,

                ))
        
        return entities
    
    def _extract_symptoms(self, text: str) -> List[ClinicalEntity]:

        doc = self.nlp(text)

                "pain", "ache", "fever", "cough", "nausea",

                    text=ent.text,

    async def generate_soap_note(

- Procedures: {[p.text for p in procedures]}

SUBJECTIVE:

OBJECTIVE:
[Physical exam findings, vital signs, test results]

ASSESSMENT:

PLAN:

Output as JSON:

    "plan": "..."

            response = await self.client.chat.completions.create(
                model=self.model,
                messages=[
                    {"role": "system", "content": system_prompt},
                    {"role": "user", "content": user_prompt}
                ],

                objective=parsed.get("objective", ""),
                assessment=parsed.get("assessment", ""),
                plan=parsed.get("plan", ""),
                entities=entities,
                confidence_score=self._calculate_confidence(entities)
            )
            
            logger.info(

                model=self.model,
                confidence=soap.confidence_score

            return soap
            
        except Exception as e:
            logger.error("soap_generation_failed", error=str(e))
            # Return fallback SOAP note
            return self._fallback_soap(transcript, entities)
    
    def _calculate_confidence(self, entities: List[ClinicalEntity]) -> float:
        """Calculate overall confidence based on entity extraction."""
        if not entities:
            return 0.5
        
        avg_conf = sum(e.confidence for e in entities) / len(entities)
        # Boost confidence if we have structured codes
        code_bonus = 0.1 if any(e.icd10 for e in entities) else 0
        return min(0.95, avg_conf + code_bonus)

        self, 
        transcript: str, 
        entities: List[ClinicalEntity]
    ) -> SOAPNote:
        """Generate basic SOAP note if LLM fails."""
        return SOAPNote(
            subjective="See transcript for details.",
            objective="See transcript for details.",
            assessment="Clinical assessment based on patient presentation.",
            plan="Follow standard clinical protocols.",
            entities=entities,
            confidence_score=0.3
        )

@router.websocket("/ws/{session_id}")
async def websocket_endpoint(websocket: WebSocket, session_id: str):
    """
    WebSocket for real-time audio streaming.
    Receives audio chunks, returns live transcription.
    """
    await websocket.accept()
    logger.info("websocket_connected", session_id=session_id)
    
    audio_buffer = bytearray()
    
    try:
        while True:
            message = await websocket.receive()
            
            if "bytes" in message:
                # Audio chunk received
                audio_buffer.extend(message["bytes"])
                await websocket.send_json({
                    "type": "ack",
                    "bytes_received": len(message["bytes"])
                })
                
            elif "text" in message:
                # Control message
                data = message["text"]
                
                if data == "END_RECORDING":
                    # Process complete audio
                    await websocket.send_json({
                        "type": "processing",
                        "message": "Transcribing audio..."
                    })
                    
                    # Run ASR
                    start_time = time.time()
                    asr_result = await get_asr_pipeline().transcribe(
                        bytes(audio_buffer),
                        sample_rate=16000
                    )
                    
                    # Update session
                    await get_db().update_session(session_id, {
                        "status": SessionStatus.PROCESSING,
                        "transcript": asr_result["transcript"],
                        "audio_duration_seconds": len(audio_buffer) / 32000  # Approximate
                    })
                    
                    await websocket.send_json({
                        "type": "transcription",
                        "transcript": asr_result["transcript"],
                        "confidence": asr_result["confidence"],
                        "processing_time_ms": int((time.time() - start_time) * 1000)
                    })
                    
                    # Extract entities
                    entities = await get_nlp_pipeline().extract_entities(
                        asr_result["transcript"]
                    )
                    
                    await websocket.send_json({
                        "type": "entities",
                        "entities": [e.model_dump() for e in entities]
                    })
                    
                    # Generate SOAP note
                    soap_note = await get_llm_service().generate_soap_note(
                        transcript=asr_result["transcript"],
                        entities=entities
                    )
                    
                    # Save to DB
                    await get_db().update_session(session_id, {
                        "status": SessionStatus.COMPLETED,
                        "soap_note": soap_note.model_dump(),
                        "completed_at": datetime.utcnow()
                    })
                    
                    await get_db().save_soap_note(session_id, soap_note)
                    
                    await websocket.send_json({
                        "type": "soap_note",
                        "soap_note": soap_note.model_dump()
                    })
                    
                    break
                    
    except WebSocketDisconnect:
        logger.info("websocket_disconnected", session_id=session_id)
        await get_db().update_session(session_id, {
            "status": SessionStatus.ERROR
        })
    except Exception as e:
        logger.error("websocket_error", session_id=session_id, error=str(e))
        await websocket.send_json({
            "type": "error",
            "message": str(e)
        })

"""High-performance rule engine for prior authorization."""
from typing import List, Dict, Any, Tuple
from app.models.policy import (
    PayerPolicy, PolicyRule, PriorAuthRequest, 
    PriorAuthDecision, Decision
)
import numpy as np
from datetime import datetime
import re
import structlog

logger = structlog.get_logger()


class RuleEngine:
    """
    Forward-chaining rule engine for prior auth decisions.
    Replaces Drools with Python-native implementation for easier deployment.
    """
    
    def __init__(self):
        self.policies: Dict[str, PayerPolicy] = {}
        self.logger = structlog.get_logger()
    
    def load_policy(self, policy: PayerPolicy):
        """Load a policy into the engine."""
        self.policies[policy.policy_id] = policy
        self.logger.info("policy_loaded", policy_id=policy.policy_id)
    
    def evaluate_request(
        self, 
        request: PriorAuthRequest
    ) -> PriorAuthDecision:
        """
        Evaluate a prior auth request against all matching policies.
        
        Returns decision with confidence score and explanation.
        """
        # Find matching policies
        matching_policies = self._find_matching_policies(request)
        
        if not matching_policies:
            # No policy found - escalate with suggestions
            return PriorAuthDecision(
                request_id=request.request_id,
                decision=Decision.ESCALATE,
                confidence_score=0.0,
                reasoning="No matching policy found for procedure/diagnosis combination",
                suggested_diagnosis_codes=self._suggest_diagnoses(request)
            )
        
        # Evaluate against best matching policy
        best_policy = matching_policies[0]
        score, matched_rules, reasoning = self._evaluate_policy(
            request, best_policy
        )
        
        # Determine decision based on thresholds
        if score >= best_policy.auto_approve_threshold:
            decision = Decision.AUTO_APPROVE
        elif score <= best_policy.auto_deny_threshold:
            decision = Decision.AUTO_DENY
        else:
            decision = Decision.ESCALATE
        
        return PriorAuthDecision(
            request_id=request.request_id,
            decision=decision,
            confidence_score=round(score, 3),
            matched_policy_id=best_policy.policy_id,
            matched_rules=matched_rules,
            reasoning=reasoning,
            required_documents=self._get_required_docs(request, best_policy),
            estimated_decision_time_hours=48 if decision == Decision.ESCALATE else 0
        )
    
    def _find_matching_policies(
        self, 
        request: PriorAuthRequest
    ) -> List[PayerPolicy]:
        """Find policies matching procedure and diagnosis."""
        matches = []
        
        for policy in self.policies.values():
            # Check procedure code match
            proc_match = (
                request.procedure_code in policy.procedure_codes or
                any(request.procedure_code.startswith(p) for p in policy.procedure_codes)
            )
            
            # Check diagnosis match (any code matches)
            diag_match = any(
                d in policy.diagnosis_codes for d in request.diagnosis_codes
            )
            
            # Check payer match
            payer_match = request.payer_id == policy.payer_id
            
            if proc_match and diag_match and payer_match:
                matches.append(policy)
        
        # Sort by specificity (most specific first)
        matches.sort(key=lambda p: len(p.procedure_codes) + len(p.diagnosis_codes))
        return matches
    
    def _evaluate_policy(
        self, 
        request: PriorAuthRequest, 
        policy: PayerPolicy
    ) -> Tuple[float, List[str], str]:
        """
        Evaluate request against policy rules.
        
        Returns: (confidence_score, matched_rules, reasoning)
        """
        rule_scores = []
        matched_rules = []
        reasoning_parts = []
        
        context = {
            "procedure_code": request.procedure_code,
            "diagnosis_codes": request.diagnosis_codes,
            "age": request.patient_age,
            "gender": request.patient_gender,
            "previous_procedures": request.previous_procedures,
            "current_medications": request.current_medications,
        }
        
        for rule in policy.rules:
            score = self._evaluate_rule(rule, context)
            weighted_score = score * rule.weight
            rule_scores.append(weighted_score)
            
            if score > 0.5:
                matched_rules.append(f"{rule.field}:{rule.operator}:{rule.value}")
                reasoning_parts.append(
                    f"{rule.field} {rule.operator} {rule.value}: {score:.2f}"
                )
        
        # Calculate harmonic mean of rule scores (punishes missing rules)
        if rule_scores:
            confidence = len(rule_scores) / sum(1/s if s > 0 else 1 for s in rule_scores)
            confidence = min(1.0, confidence / len(rule_scores))
        else:
            confidence = 0.0
        
        # Medical necessity check
        if policy.medical_necessity:
            med_nec_score = self._check_medical_necessity(
                request, policy.medical_necessity
            )
            confidence *= med_nec_score
        
        reasoning = f"Policy {policy.policy_id}: " + "; ".join(reasoning_parts)
        
        return confidence, matched_rules, reasoning
    
    def _evaluate_rule(
        self, 
        rule: PolicyRule, 
        context: Dict[str, Any]
    ) -> float:
        """Evaluate a single rule against context."""
        field_value = context.get(rule.field)
        
        if field_value is None:
            return 0.0
        
        try:
            if rule.operator == "eq":
                return 1.0 if str(field_value) == str(rule.value) else 0.0
            elif rule.operator == "ne":
                return 1.0 if str(field_value) != str(rule.value) else 0.0
            elif rule.operator == "in":
                return 1.0 if field_value in rule.value else 0.0
            elif rule.operator == "contains":
                return 1.0 if rule.value in field_value else 0.0
            elif rule.operator == "gt":
                return 1.0 if float(field_value) > float(rule.value) else 0.0
            elif rule.operator == "lt":
                return 1.0 if float(field_value) < float(rule.value) else 0.0
            elif rule.operator == "gte":
                return 1.0 if float(field_value) >= float(rule.value) else 0.0
            elif rule.operator == "lte":
                return 1.0 if float(field_value) <= float(rule.value) else 0.0
        except (ValueError, TypeError):
            return 0.0
        
        return 0.0
    
    def _check_medical_necessity(
        self, 
        request: PriorAuthRequest,
        criteria: Any
    ) -> float:
        """Check medical necessity from clinical notes."""
        if not request.clinical_notes:
            return 0.5  # Neutral if no notes
        
        notes_lower = request.clinical_notes.lower()
        score = 0.5
        
        # Check for evidence keywords
        evidence_keywords = {
            "failed": 0.2,
            "conservative": 0.2,
            "therapy": 0.1,
            "medication": 0.1,
            "tried": 0.1,
            "weeks": 0.1,
            "months": 0.1,
            "pain": 0.1,
            "severe": 0.1,
        }
        
        for keyword, weight in evidence_keywords.items():
            if keyword in notes_lower:
                score += weight
        
        return min(1.0, score)
    
    def _suggest_diagnoses(self, request: PriorAuthRequest) -> List[str]:
        """Suggest alternative diagnosis codes."""
        # Simplified - in production use mapping tables
        suggestions = []
        proc = request.procedure_code
        
        # Common mappings
        mappings = {
            "73706": ["M17.11", "M17.12", "M25.561"],  # MRI knee -> OA
            "71020": ["J44.1", "J18.9"],  # Chest X-ray -> COPD, Pneumonia
            "80053": ["E11.9", "I10"],  # CMP -> Diabetes, HTN
        }
        
        if proc in mappings:
            suggestions = mappings[proc]
        
        return suggestions
    
    def _get_required_docs(
        self, 
        request: PriorAuthRequest, 
        policy: PayerPolicy
    ) -> List[str]:
        """Determine required documentation for escalation."""
        docs = ["Clinical notes", "Imaging reports"]
        
        if policy.medical_necessity:
            docs.extend(policy.medical_necessity.required_evidence)
        
        return docs

/**
 * Real-time voice recorder with WebRTC
 * Streams audio to backend WebSocket for transcription
 */
import { useState, useRef, useCallback } from 'react';
import { Mic, Square, Loader2 } from 'lucide-react';
import { clsx, type ClassValue } from 'clsx';
import { twMerge } from 'tailwind-merge';

function cn(...inputs: ClassValue[]) {
  return twMerge(clsx(inputs));
}

interface VoiceRecorderProps {
  sessionId: string;
  onTranscript: (transcript: string) => void;
  onSoapNote: (note: any) => void;
  onStatusChange?: (status: 'idle' | 'recording' | 'processing' | 'completed' | 'error') => void;
}

export function VoiceRecorder({ 
  sessionId, 
  onTranscript, 
  onSoapNote,
  onStatusChange 
}: VoiceRecorderProps) {
  const [status, setStatus] = useState<'idle' | 'recording' | 'processing' | 'completed' | 'error'>('idle');
  const [recordingTime, setRecordingTime] = useState(0);
  const [error, setError] = useState<string | null>(null);
  
  const mediaRecorderRef = useRef<MediaRecorder | null>(null);
  const wsRef = useRef<WebSocket | null>(null);
  const timerRef = useRef<NodeJS.Timeout | null>(null);
  const audioChunksRef = useRef<Blob[]>([]);

  const updateStatus = useCallback((newStatus: typeof status) => {
    setStatus(newStatus);
    onStatusChange?.(newStatus);
  }, [onStatusChange]);

  const startRecording = async () => {
    try {
      // Get microphone access
      const stream = await navigator.mediaDevices.getUserMedia({ 
        audio: {
          sampleRate: 16000,
          channelCount: 1,
          echoCancellation: true,
          noiseSuppression: true
        } 
      });

      // Setup WebSocket connection
      const ws = new WebSocket(`ws://localhost:8000/api/v1/ws/${sessionId}`);
      wsRef.current = ws;

      ws.onopen = () => {
        console.log('WebSocket connected');
      };

      ws.onmessage = (event) => {
        const data = JSON.parse(event.data);
        handleWebSocketMessage(data);
      };

      ws.onerror = (err) => {
        console.error('WebSocket error:', err);
        setError('Connection error');
        updateStatus('error');
      };

      // Setup MediaRecorder
      const mediaRecorder = new MediaRecorder(stream, {
        mimeType: 'audio/webm;codecs=opus'
      });
      mediaRecorderRef.current = mediaRecorder;

      mediaRecorder.ondataavailable = (event) => {
        if (event.data.size > 0) {
          audioChunksRef.current.push(event.data);
          // Send to WebSocket in real-time (optional: buffer and send chunks)
          if (ws.readyState === WebSocket.OPEN) {
            event.data.arrayBuffer().then(buffer => {
              ws.send(buffer);
            });
          }
        }
      };

      mediaRecorder.start(1000); // Collect data every second
      updateStatus('recording');
      
      // Start timer
      timerRef.current = setInterval(() => {
        setRecordingTime(prev => prev + 1);
      }, 1000);

    } catch (err) {
      console.error('Failed to start recording:', err);
      setError('Microphone access denied');
      updateStatus('error');
    }
  };

  const stopRecording = () => {
    if (mediaRecorderRef.current && mediaRecorderRef.current.state !== 'inactive') {
      mediaRecorderRef.current.stop();
      mediaRecorderRef.current.stream.getTracks().forEach(track => track.stop());
      
      // Send end signal
      if (wsRef.current?.readyState === WebSocket.OPEN) {
        wsRef.current.send('END_RECORDING');
      }
    }

    if (timerRef.current) {
      clearInterval(timerRef.current);
    }

    updateStatus('processing');
  };

  const handleWebSocketMessage = (data: any) => {
    switch (data.type) {
      case 'transcription':
        onTranscript(data.transcript);
        break;
      case 'entities':
        console.log('Entities extracted:', data.entities);
        break;
      case 'soap_note':
        onSoapNote(data.soap_note);
        updateStatus('completed');
        wsRef.current?.close();
        break;
      case 'error':
        setError(data.message);
        updateStatus('error');
        break;
    }
  };

  const formatTime = (seconds: number) => {
    const mins = Math.floor(seconds / 60);
    const secs = seconds % 60;
    return `${mins}:${secs.toString().padStart(2, '0')}`;
  };

  return (
    <div className="flex flex-col items-center gap-4 p-6 bg-white rounded-lg shadow-sm border">
      <div className="flex items-center gap-4">
        <button
          onClick={status === 'recording' ? stopRecording : startRecording}
          disabled={status === 'processing' || status === 'completed'}
          className={cn(
            "relative flex items-center justify-center w-16 h-16 rounded-full transition-all",
            status === 'recording' 
              ? "bg-red-500 hover:bg-red-600 animate-pulse"
              : "bg-blue-600 hover:bg-blue-700",
            (status === 'processing' || status === 'completed') && "opacity-50 cursor-not-allowed"
          )}
        >
          {status === 'recording' ? (
            <Square className="w-6 h-6 text-white" />
          ) : status === 'processing' ? (
            <Loader2 className="w-6 h-6 text-white animate-spin" />
          ) : (
            <Mic className="w-6 h-6 text-white" />
          )}
          
          {/* Recording indicator ring */}
          {status === 'recording' && (
            <span className="absolute inline-flex h-full w-full rounded-full bg-red-400 opacity-20 animate-ping" />
          )}
        </button>

        <div className="text-left">
          <p className="text-lg font-semibold text-gray-900">
            {status === 'idle' && 'Ready to Record'}
            {status === 'recording' && `Recording... ${formatTime(recordingTime)}`}
            {status === 'processing' && 'Processing with AI...'}
            {status === 'completed' && 'Documentation Complete'}
            {status === 'error' && 'Error Occurred'}
          </p>
          <p className="text-sm text-gray-500">
            {status === 'idle' && 'Click to start voice documentation'}
            {status === 'recording' && 'Speak clearly into your microphone'}
            {status === 'processing' && 'Transcribing and generating SOAP note...'}
            {status === 'completed' && 'Review the generated note below'}
            {status === 'error' && error}
          </p>
        </div>
      </div>

      {/* Progress bar for processing */}
      {status === 'processing' && (
        <div className="w-full max-w-md">
          <div className="h-2 bg-gray-200 rounded-full overflow-hidden">
            <div className="h-full bg-blue-600 animate-[shimmer_2s_infinite] w-1/2" />
          </div>
          <p className="text-xs text-center text-gray-500 mt-2">
            WhisperX ASR → BioClinicalBERT NER → Llama 3 SOAP Generation
          </p>
        </div>
      )}
    </div>
  );
}

-- FHIR R4 compatible schema for AI-HOS
-- HIPAA-compliant audit logging

-- Enable UUID extension
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- Patients table (de-identified for analytics)
CREATE TABLE patients (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    fhir_id VARCHAR(64) UNIQUE NOT NULL,
    -- Pseudonymized identifiers
    mrn_hash VARCHAR(64) NOT NULL,
    -- Demographics (for analytics)
    birth_year INTEGER CHECK (birth_year > 1900 AND birth_year < 2100),
    gender VARCHAR(10) CHECK (gender IN ('male', 'female', 'other', 'unknown')),
    zip3 VARCHAR(3), -- First 3 digits of ZIP for geographic analysis
    -- Metadata
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    -- Consent flags
    consent_research BOOLEAN DEFAULT FALSE,
    consent_sharing BOOLEAN DEFAULT FALSE
);

-- Encounters/Visits
CREATE TABLE encounters (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    fhir_id VARCHAR(64) UNIQUE NOT NULL,
    patient_id UUID REFERENCES patients(id) ON DELETE CASCADE,
    encounter_type VARCHAR(50), -- ambulatory, emergency, inpatient
    start_time TIMESTAMP WITH TIME ZONE,
    end_time TIMESTAMP WITH TIME ZONE,
    provider_id VARCHAR(64),
    department VARCHAR(100),
    -- For readmission prediction
    discharge_disposition VARCHAR(50),
    length_of_stay_hours INTEGER,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Prior Authorization Requests
CREATE TABLE prior_auth_requests (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    request_id VARCHAR(64) UNIQUE NOT NULL,
    patient_id UUID REFERENCES patients(id),
    provider_id VARCHAR(64) NOT NULL,
    payer_id VARCHAR(50) NOT NULL,
    -- Clinical
    procedure_code VARCHAR(20) NOT NULL,
    procedure_description TEXT,
    diagnosis_codes TEXT[], -- Array of ICD-10
    -- Decision
    decision VARCHAR(20) CHECK (decision IN ('auto_approve', 'auto_deny', 'escalate', 'manual_approve', 'manual_deny')),
    confidence_score DECIMAL(3,2),
    matched_policy_id VARCHAR(64),
    -- Timing
    submitted_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    decided_at TIMESTAMP WITH TIME ZONE,
    decision_reasoning TEXT,
    -- Audit
    processed_by VARCHAR(64), -- AI service or user ID
    escalation_reason TEXT
);

-- Audit logs (HIPAA requirement)
CREATE TABLE audit_logs (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    timestamp TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    user_id VARCHAR(64) NOT NULL,
    user_role VARCHAR(50) NOT NULL,
    action VARCHAR(50) NOT NULL, -- CREATE, READ, UPDATE, DELETE
    resource_type VARCHAR(50) NOT NULL, -- Patient, Encounter, etc.
    resource_id VARCHAR(64) NOT NULL,
    purpose_of_use VARCHAR(50), -- TREATMENT, RESEARCH, PAYMENT
    consent_verified BOOLEAN DEFAULT FALSE,
    ip_address INET,
    user_agent TEXT,
    success BOOLEAN DEFAULT TRUE,
    error_message TEXT,
    -- For detailed tracking
    request_payload JSONB,
    changes_made JSONB
);

-- Indexes for performance
CREATE INDEX idx_patients_mrn_hash ON patients(mrn_hash);
CREATE INDEX idx_encounters_patient ON encounters(patient_id);
CREATE INDEX idx_encounters_start ON encounters(start_time);
CREATE INDEX idx_prior_auth_patient ON prior_auth_requests(patient_id);
CREATE INDEX idx_prior_auth_decision ON prior_auth_requests(decision);
CREATE INDEX idx_audit_user ON audit_logs(user_id, timestamp);
CREATE INDEX idx_audit_resource ON audit_logs(resource_type, resource_id);

-- Partition audit logs by month (for performance at scale)
CREATE TABLE audit_logs_2024_01 PARTITION OF audit_logs
    FOR VALUES FROM ('2024-01-01') TO ('2024-02-01');

-- Trigger for updated_at
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ language 'plpgsql';

CREATE TRIGGER update_patients_updated_at BEFORE UPDATE ON patients
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
